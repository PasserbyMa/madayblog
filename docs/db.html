2. Oracle SQL 핵심

기본 조회 & 페이징

-- 기본 조회
SELECT name, salary, department
FROM employees
WHERE department = '개발팀'
ORDER BY salary DESC;

-- 페이징 (Oracle 12c 이상)
SELECT * FROM posts
ORDER BY created_at DESC
OFFSET 0 ROWS FETCH NEXT 10 ROWS ONLY;   -- 1~10번째
OFFSET 10 ROWS FETCH NEXT 10 ROWS ONLY;  -- 11~20번째

-- 페이징 (Oracle 11g 이하 — ROWNUM 방식)
SELECT * FROM (
  SELECT ROWNUM rn, e.*
  FROM employees e
  WHERE ROWNUM <= 20
)
WHERE rn >= 11;  -- 11~20번째

-- 현재 유저 / 테이블 목록
SELECT USER FROM dual;
SELECT table_name FROM user_tables;

Oracle 전용 함수

-- NVL: NULL이면 대체값 반환 (MySQL의 IFNULL)
SELECT NVL(salary, 0) FROM employees;
SELECT NVL(category, '미분류') FROM posts;

-- DECODE: 조건 분기 (CASE WHEN 단순 버전)
SELECT DECODE(grade, 'A', '우수', 'B', '보통', '기타') FROM students;

-- TO_DATE / TO_CHAR: 날짜 변환
SELECT * FROM orders WHERE order_date >= TO_DATE('2024-01-01', 'YYYY-MM-DD');
SELECT TO_CHAR(SYSDATE, 'YYYY-MM-DD HH24:MI:SS') FROM dual;

-- SUBSTR: 문자열 자르기
SELECT SUBSTR('홍길동', 1, 1) FROM dual;  -- '홍'

-- CASE WHEN
SELECT name,
  CASE
    WHEN salary >= 5000000 THEN '고액'
    WHEN salary >= 3000000 THEN '중간'
    ELSE '일반'
  END AS salary_grade
FROM employees;

Oracle JOIN

-- INNER JOIN: 양쪽 다 있는 것만
SELECT u.name, o.product, o.amount
FROM users u
INNER JOIN orders o ON u.id = o.user_id;

-- LEFT JOIN: 왼쪽 전부 + 오른쪽은 있으면 포함 (없으면 NULL)
SELECT u.name, o.product
FROM users u
LEFT JOIN orders o ON u.id = o.user_id;

-- 3테이블 JOIN
SELECT u.name, o.product, p.price
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
LEFT JOIN products p ON o.product_id = p.id;

Oracle GROUP BY + HAVING

-- 부서별 인원수, 평균 급여 (300만 이상 부서만)
SELECT department, COUNT(*) AS cnt, AVG(salary) AS avg_sal
FROM employees
WHERE status = 'Y'            -- GROUP BY 전 필터 (재직자만)
GROUP BY department
HAVING AVG(salary) >= 3000000 -- GROUP BY 후 필터 (집계 결과에 조건)
ORDER BY avg_sal DESC;

Oracle 시퀀스 (AUTO_INCREMENT 대체)

-- 시퀀스 생성
CREATE SEQUENCE posts_seq START WITH 1 INCREMENT BY 1 NOCACHE;

-- 테이블에 적용 (Oracle 12c+)
CREATE TABLE posts (
    id NUMBER DEFAULT posts_seq.NEXTVAL PRIMARY KEY,
    ...
);

-- 현재 시퀀스 값 확인
SELECT posts_seq.CURRVAL FROM dual;

JOIN 예시 (MySQL)

-- INNER JOIN: 가입현황(subscription) + 고객(member) + 요금제(plan)
SELECT m.name, p.plan_name, p.monthly_fee, s.status, s.start_date
FROM subscription s
JOIN member m ON s.member_id = m.id
JOIN plan   p ON s.plan_id   = p.id
WHERE s.status = 'ACTIVE'
ORDER BY s.start_date DESC;

-- LEFT JOIN: 가입 안 한 고객도 포함
SELECT m.name, s.status
FROM member m
LEFT JOIN subscription s ON m.id = s.member_id;

MySQL 테이블 생성 (DDL)

-- 데이터베이스 생성 & 선택
CREATE DATABASE v3db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE v3db;

-- AUTO_INCREMENT, 외래키, 제약조건
CREATE TABLE member (
    id         BIGINT AUTO_INCREMENT PRIMARY KEY,
    name       VARCHAR(50)  NOT NULL,
    email      VARCHAR(100) NOT NULL UNIQUE,
    phone      VARCHAR(20),
    birth_date DATE,
    status     VARCHAR(10)  NOT NULL DEFAULT 'ACTIVE',
    created_at DATETIME     DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE plan (
    id          BIGINT AUTO_INCREMENT PRIMARY KEY,
    plan_name   VARCHAR(50)  NOT NULL,
    monthly_fee INT          NOT NULL,
    description VARCHAR(200)
);

CREATE TABLE subscription (
    id         BIGINT AUTO_INCREMENT PRIMARY KEY,
    member_id  BIGINT NOT NULL,
    plan_id    BIGINT NOT NULL,
    status     VARCHAR(10) DEFAULT 'ACTIVE',
    start_date DATE,
    FOREIGN KEY (member_id) REFERENCES member(id),
    FOREIGN KEY (plan_id)   REFERENCES plan(id)
);

-- 초기 데이터 삽입 (INSERT IGNORE: 중복 시 에러 무시 → 재실행 가능)
INSERT IGNORE INTO plan (id, plan_name, monthly_fee) VALUES
    (1, 'Basic',    9900),
    (2, 'Standard', 14900),
    (3, 'Premium',  19900);

-- ON DUPLICATE KEY UPDATE: 중복이면 업데이트, 없으면 삽입
INSERT INTO member (email, name, status)
VALUES ('hong@test.com', '홍길동', 'ACTIVE')
ON DUPLICATE KEY UPDATE name='홍길동', status='ACTIVE';

4. 고급 SQL (서브쿼리 / GROUP BY+HAVING / 윈도우함수 / CTE)

서브쿼리 (Subquery)

-- 평균보다 비싼 요금제에 가입한 고객
SELECT name, email FROM member
WHERE id IN (
    SELECT member_id FROM subscription s
    JOIN plan p ON s.plan_id = p.id
    WHERE p.monthly_fee > (SELECT AVG(monthly_fee) FROM plan)
);

-- NOT EXISTS: 가입이 없는 고객 조회
SELECT name FROM member m
WHERE NOT EXISTS (
    SELECT 1 FROM subscription s WHERE s.member_id = m.id
);

-- 도서관 테마: 평균 가격보다 비싼 책
SELECT title, price FROM book
WHERE price > (SELECT AVG(price) FROM book);
IN vs EXISTS 차이
IN: 서브쿼리 결과를 전부 메모리에 올림. 결과 집합이 클 때 느림.
EXISTS: 일치하는 행 하나만 찾으면 즉시 종료(early termination). 대용량에서 더 빠름.

GROUP BY + HAVING

-- 요금제별 가입자 수, 2명 이상인 것만
SELECT p.plan_name, COUNT(*) AS subscriber_count
FROM subscription s
JOIN plan p ON s.plan_id = p.id
GROUP BY p.plan_name
HAVING subscriber_count >= 2
ORDER BY subscriber_count DESC;

-- 부서별 평균 급여 300만 이상
SELECT department, COUNT(*) AS cnt, AVG(salary) AS avg_sal
FROM employees
WHERE status = 'Y'             -- WHERE: 행 단위 필터 (GROUP BY 전)
GROUP BY department
HAVING AVG(salary) >= 3000000 -- HAVING: 그룹 단위 필터 (GROUP BY 후)
ORDER BY avg_sal DESC;

-- 도서관 테마: 저자별 판매액 집계
SELECT a.name, COUNT(*) AS book_count, SUM(b.price * o.quantity) AS total_sales
FROM orders o
JOIN book b ON o.book_id = b.id
JOIN author a ON b.author_id = a.id
GROUP BY a.name
HAVING total_sales > 20000
ORDER BY total_sales DESC;
WHERE vs HAVING
WHERE — GROUP BY 이전에 행 필터링 (집계 함수 사용 불가)
HAVING — GROUP BY 이후에 그룹 필터링 (COUNT, SUM 등 집계 함수 사용 가능)

윈도우 함수 (Window Function)

-- 요금제별 월정액 + 전체 대비 순위
SELECT m.name, p.plan_name, p.monthly_fee,
       RANK() OVER (ORDER BY p.monthly_fee DESC) AS fee_rank
FROM subscription s
JOIN member m ON s.member_id = m.id
JOIN plan p ON s.plan_id = p.id;

-- 독자별 지출 합계 + 순위 (도서관 테마)
SELECT r.name, SUM(b.price * o.quantity) AS spent,
       RANK() OVER (ORDER BY SUM(b.price * o.quantity) DESC) AS rnk
FROM orders o
JOIN reader r ON o.reader_id = r.id
JOIN book b ON o.book_id = b.id
GROUP BY r.name;

-- 전체 대비 비율
SELECT plan_name, cnt,
       ROUND(cnt * 100.0 / SUM(cnt) OVER (), 1) AS pct
FROM (SELECT plan_name, COUNT(*) AS cnt
      FROM subscription JOIN plan ON subscription.plan_id = plan.id
      GROUP BY plan_name) sub;
함수설명예시
RANK()순위 (동순위 있으면 번호 건너뜀 1,1,3)RANK() OVER (ORDER BY salary DESC)
DENSE_RANK()순위 (동순위도 연속 1,1,2)DENSE_RANK() OVER (ORDER BY score DESC)
ROW_NUMBER()고유 순번 (동순위도 각각 다른 번호)ROW_NUMBER() OVER (ORDER BY id)
SUM() OVER ()전체 합을 각 행에 붙임SUM(cnt) OVER ()
SUM() OVER (PARTITION BY)그룹별 합을 각 행에 붙임SUM(salary) OVER (PARTITION BY dept)
LAG(col, n)n번째 이전 행 값LAG(salary, 1) OVER (ORDER BY id)
LEAD(col, n)n번째 다음 행 값LEAD(salary, 1) OVER (ORDER BY id)
GROUP BY vs 윈도우 함수 차이
GROUP BY: 행을 합쳐서 수를 줄임 (n개 → 그룹 수).
윈도우 함수: 행 수는 그대로 유지하면서 집계값을 각 행에 추가.

CTE (Common Table Expression — WITH 절)

-- 기본 CTE: 서브쿼리를 이름 붙인 임시 결과로 분리
WITH plan_revenue AS (
    SELECT p.plan_name, SUM(p.monthly_fee) AS revenue
    FROM subscription s
    JOIN plan p ON s.plan_id = p.id
    WHERE s.status = 'ACTIVE'
    GROUP BY p.plan_name
)
SELECT * FROM plan_revenue ORDER BY revenue DESC;

-- 여러 CTE 연결
WITH
  top_readers AS (
      SELECT reader_id, SUM(b.price * o.quantity) AS spent
      FROM orders o JOIN book b ON o.book_id = b.id
      GROUP BY reader_id
  ),
  reader_info AS (
      SELECT r.name, r.email, t.spent
      FROM reader r JOIN top_readers t ON r.id = t.reader_id
  )
SELECT * FROM reader_info WHERE spent > 50000 ORDER BY spent DESC;

-- 도서관 테마: 저자 매출 TOP
WITH author_sales AS (
    SELECT a.name, SUM(b.price * o.quantity) AS total
    FROM orders o
    JOIN book b ON o.book_id = b.id
    JOIN author a ON b.author_id = a.id
    GROUP BY a.name
)
SELECT * FROM author_sales WHERE total > 15000;
서브쿼리 vs CTE
기능은 같지만 CTE가 훨씬 읽기 쉬움. 같은 집계를 여러 번 참조할 때 CTE가 유리.
지원: MariaDB 10.2+, MySQL 8.0+, Oracle 9i+, PostgreSQL 8.4+

전체 SELECT 실행 순서

FROM → JOIN → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT

FROM      테이블 선택
JOIN      테이블 연결
WHERE     행 단위 필터 (집계 함수 사용 불가)
GROUP BY  그룹화
HAVING    그룹 단위 필터 (집계 함수 사용 가능)
SELECT    컬럼/집계 계산
ORDER BY  정렬
LIMIT     개수 제한

5. 인덱스 전략

검색 속도를 높이는 자료구조. 책의 목차처럼 원하는 데이터 위치를 바로 찾아감.

기본 개념

상황전략
자주 조회하는 컬럼 (WHERE, JOIN ON)단일 인덱스
복합 조건 (WHERE a=? AND b=?)복합 인덱스 (선택도 높은 컬럼 먼저)
PK 컬럼자동으로 클러스터드 인덱스 생성
INSERT/UPDATE/DELETE 많은 컬럼인덱스 최소화 (인덱스 유지 비용)
카디널리티가 낮은 컬럼 (성별 등)인덱스 효과 없음, 피하는 것이 좋음

인덱스 생성 & 확인

-- 단일 인덱스
CREATE INDEX idx_member_email ON member(email);

-- 복합 인덱스 (자주 함께 조회되는 컬럼)
CREATE INDEX idx_subscription_member_status ON subscription(member_id, status);

-- 인덱스 사용 여부 확인 (EXPLAIN)
EXPLAIN SELECT * FROM member WHERE email = 'test@test.com';
-- key 컬럼에 idx_member_email이 나오면 인덱스 사용
-- type이 'ALL'이면 전체 스캔 → 인덱스 없거나 효과 없음

-- 인덱스 목록 확인
SHOW INDEX FROM member;

-- 인덱스 삭제
DROP INDEX idx_member_email ON member;
인덱스 단점
조회는 빠르지만 INSERT/UPDATE/DELETE 시 인덱스도 갱신해야 해서 쓰기가 느려짐.
조회가 많은 컬럼에만 선별적으로 사용할 것.

6. 트랜잭션 & ACID

트랜잭션 = 여러 DB 작업을 하나의 단위로 묶음. 전부 성공하거나 전부 실패(롤백).

ACID

속성의미예시
Atomicity (원자성)전부 성공 or 전부 실패. 중간 없음.이체: A -10만, B +10만 중 하나 실패 → 둘 다 취소
Consistency (일관성)트랜잭션 전후 DB 규칙(제약조건) 항상 유지외래키 제약조건 위반 시 커밋 불가
Isolation (격리성)동시 실행 트랜잭션이 서로 영향 안 줌A가 수정 중인 행을 B가 못 읽음
Durability (지속성)커밋된 데이터는 장애가 나도 사라지지 않음서버 다운돼도 커밋된 주문은 유지

MySQL 트랜잭션 SQL

-- 명시적 트랜잭션
START TRANSACTION;

UPDATE member SET status='INACTIVE' WHERE id=1;
DELETE FROM subscription WHERE member_id=1;

COMMIT;    -- 성공 시 확정
ROLLBACK;  -- 실패 시 취소 (START TRANSACTION 시점으로 되돌림)

Spring @Transactional

// @Transactional: AOP 프록시가 감싸서 메서드 실행 전 BEGIN, 이후 COMMIT/ROLLBACK
@Transactional
public void deleteMemberWithSubscriptions(Long id) {
    subscriptionMapper.deleteByMemberId(id);
    memberMapper.delete(id);
    // 중간에 예외 발생 시 두 삭제 모두 ROLLBACK
}

// RuntimeException(unchecked) → 자동 롤백
// Checked Exception → 기본적으로 롤백 안 됨 (rollbackFor 옵션으로 지정 가능)
@Transactional(rollbackFor = Exception.class)

7. MongoDB CRUD & Aggregate

기본 구조

SQL              MongoDB
테이블(Table)  →  컬렉션(Collection)
행(Row)        →  도큐먼트(Document)  {JSON 형태}
컬럼(Column)   →  필드(Field)
PRIMARY KEY    →  _id (자동 생성)

기본 CRUD

// 조회 (SELECT)
db.users.find({ age: { $gte: 20 } })           // age >= 20
db.users.find({ name: "홍길동" }, { _id: 0 })  // 특정 필드만 (projection)
db.users.findOne({ email: "test@test.com" })   // 한 건만

// 삽입 (INSERT)
db.users.insertOne({ name: "홍길동", age: 30, skills: ["Java", "Spring"] })
db.users.insertMany([{ name: "A" }, { name: "B" }])

// 수정 (UPDATE)
db.users.updateOne(
  { name: "홍길동" },
  { $set: { age: 31 } }   // $set: 특정 필드만 수정
)
db.users.updateMany({ status: "inactive" }, { $set: { active: false } })

// 삭제 (DELETE)
db.users.deleteOne({ name: "홍길동" })
db.users.deleteMany({ age: { $lt: 18 } })

비교 연산자

$eq  → 같음        { age: { $eq: 30 } }
$ne  → 다름        { age: { $ne: 30 } }
$gt  → 초과        { age: { $gt: 20 } }
$gte → 이상        { age: { $gte: 20 } }
$lt  → 미만        { age: { $lt: 30 } }
$lte → 이하        { age: { $lte: 30 } }
$in  → 포함        { age: { $in: [20, 30] } }
$and → 그리고      { $and: [{ age: { $gte: 20 } }, { status: "active" }] }
$or  → 또는        { $or: [{ age: 20 }, { age: 30 }] }

Aggregate (집계 — GROUP BY 상당)

// 부서별 평균 나이 (파이프라인 방식)
db.users.aggregate([
    { $match: { status: "active" } },        // WHERE (필터)
    { $group: {
        _id: "$department",                  // GROUP BY department
        avgAge: { $avg: "$age" },            // AVG(age)
        count: { $sum: 1 }                   // COUNT(*)
    }},
    { $sort: { avgAge: -1 } },              // ORDER BY avgAge DESC
    { $limit: 5 }                            // LIMIT 5
]);

// 주요 집계 연산자
$sum   → 합계
$avg   → 평균
$min   → 최솟값
$max   → 최댓값
$count → 개수 (또는 $sum: 1)

8. DynamoDB 구조 & 쿼리

기본 구조

테이블: users
┌────────────────┬────────────┬──────────────┐
│ userId (PK)    │ createdAt  │ name         │
│ Partition Key  │ Sort Key   │              │
├────────────────┼────────────┼──────────────┤
│ user_001       │ 2024-01-01 │ 홍길동       │
│ user_001       │ 2024-02-01 │ 홍길동       │  ← 같은 PK, 다른 SK
│ user_002       │ 2024-01-15 │ 김철수       │
└────────────────┴────────────┴──────────────┘

Partition Key (필수): 데이터를 어느 서버에 저장할지 결정 (샤딩)
Sort Key (선택): 같은 PK 안에서 정렬 / 범위 검색

CRUD (AWS SDK v3)

import { DynamoDBClient } from "@aws-sdk/client-dynamodb";
import { DynamoDBDocumentClient, GetCommand, PutCommand,
         UpdateCommand, DeleteCommand, QueryCommand } from "@aws-sdk/lib-dynamodb";

const client = new DynamoDBClient({ region: "ap-northeast-2" });
const db = DynamoDBDocumentClient.from(client);

// 조회 (PK로 정확히 한 건)
const { Item } = await db.send(new GetCommand({
    TableName: "users",
    Key: { userId: "user_001" }
}));

// 삽입
await db.send(new PutCommand({
    TableName: "users",
    Item: { userId: "user_003", name: "이영희", age: 25 }
}));

// 수정
await db.send(new UpdateCommand({
    TableName: "users",
    Key: { userId: "user_001" },
    UpdateExpression: "SET #n = :name, age = :age",
    ExpressionAttributeNames: { "#n": "name" },  // 예약어 충돌 방지
    ExpressionAttributeValues: { ":name": "홍길동", ":age": 31 }
}));

// 삭제
await db.send(new DeleteCommand({
    TableName: "users",
    Key: { userId: "user_003" }
}));

Query vs Scan

// Query: Partition Key 기준 검색 (인덱스 사용 → 빠름) ← 실무에서 이것 써야 함
await db.send(new QueryCommand({
    TableName: "orders",
    KeyConditionExpression: "userId = :uid AND createdAt BETWEEN :start AND :end",
    ExpressionAttributeValues: {
        ":uid": "user_001",
        ":start": "2024-01-01",
        ":end": "2024-12-31"
    }
}));

// Scan: 테이블 전체 순회 (느림, 비용 많이 듦 → 최대한 피할 것)
await db.send(new ScanCommand({
    TableName: "users",
    FilterExpression: "age >= :age",
    ExpressionAttributeValues: { ":age": 20 }
}));
Query vs Scan 핵심 차이
Query = Partition Key 필수, 인덱스 사용 → 빠름, 비용 효율적.
Scan = PK 없이도 되지만 전체 테이블 탐색 → 느림, 비용 많이 듦.

FilterExpression 연산자

= , <> , < , <= , > , >=          비교
BETWEEN :a AND :b                 범위
begins_with(field, :val)          문자열 시작값
contains(field, :val)             포함 여부
attribute_exists(field)           필드 존재 여부
attribute_not_exists(field)       필드 없음 여부
AND , OR , NOT                    논리 연산

FilterExpression: "age >= :age AND #s = :status"
ExpressionAttributeNames:  { "#s": "status" }   // 예약어 'status' 우회
ExpressionAttributeValues: { ":age": 20, ":status": "active" }
