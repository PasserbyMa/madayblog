3. MongoDB 연결 & 데이터 모델

ConnectMongoDB.ts — 글로벌 커넥션 캐싱

// Next.js dev 모드는 HMR로 모듈이 재실행됨 → 커넥션이 계속 새로 생기는 문제 방지
// globalThis에 캐싱해서 이미 연결된 경우 재사용
let cached = (globalThis as CustomGlobalThis).mongooseCache;
if (!cached) cached = { conn: undefined, promise: undefined };

export const ConnectMongoDB = async () => {
    if (cached.conn?.connection.readyState === 1) return cached.conn;  // 이미 연결됨
    if (cached.promise) return cached.promise;                         // 연결 진행 중

    const opts = { bufferCommands: false, authSource: 'admin' };
    cached.promise = mongoose.connect(MONGO_URI, opts);
    cached.conn = await cached.promise;
    return cached.conn;
};

포스트 조회 (페이징 + 검색)

// page: 1부터 시작, limit: 한 페이지당 개수, query: 검색어
export async function GetPostsAllData(page = 1, limit = 10, query = '') {
    await ConnectMongoDB();

    const filter = query
        ? { $or: [
            { title:    { $regex: query, $options: 'i' } },
            { category: { $regex: query, $options: 'i' } },
          ]}
        : {};

    const total = await ModelPostsSetting.countDocuments(filter);
    const posts = await ModelPostsSetting.find(filter)
        .sort({ date: -1 })
        .skip((page - 1) * limit)
        .limit(limit)
        .lean();

    // Markdown → HTML 변환
    const returnArr = await Promise.all(
        posts.map(async (p) => {
            const { contentHtml } = await ConvertMarkdownToHtml(p.content);
            return { ...p, _id: p._id.toString(), contentHtml };
        })
    );
    return { posts: returnArr, total };
}

4. GitHub OAuth & JWT 세션

OAuth 흐름

사용자 [로그인 클릭]
    ↓
GET /api/auth/login
    ↓  GitHub OAuth URL로 redirect
    └→ https://github.com/login/oauth/authorize?client_id=...&redirect_uri=...
                ↓  사용자가 GitHub에서 허용
GET /api/auth/callback/github?code=xxxxxx
    ↓
① code → access_token (GitHub API)
    ↓
② access_token → user info (https://api.github.com/user)
    ↓
③ isAdmin = (user.id === GIT_ADMIN_ID)
    ↓
④ JWT 서명 (jose, HS256, 7일 만료)
    ↓
⑤ Set-Cookie: ma_session=... (HttpOnly, Secure)
    ↓
메인 페이지로 redirect

GitHub OAuth 콜백 코드

// /api/auth/callback/github/route.ts
export async function GET(request: Request) {
    const code = new URL(request.url).searchParams.get('code');

    // ① code → access_token
    const { access_token } = await fetch('https://github.com/login/oauth/access_token', {
        method: 'POST',
        headers: { Accept: 'application/json', 'Content-Type': 'application/json' },
        body: JSON.stringify({ client_id, client_secret, code }),
    }).then(r => r.json());

    // ② access_token → user
    const user = await fetch('https://api.github.com/user', {
        headers: { Authorization: `Bearer ${access_token}`, 'User-Agent': 'madayblog' },
    }).then(r => r.json());

    // ③ JWT 발급 + 쿠키 설정
    const sessionToken = await signSession({
        id: user.id,
        name: user.name ?? user.login,
        email: user.email ?? null,
        image: user.avatar_url,
        isAdmin: user.id === Number(process.env.GIT_ADMIN_ID),  // 관리자 여부
    });

    const cookie = `ma_session=${sessionToken}; HttpOnly; Secure; Path=/; Max-Age=${7*24*3600}; SameSite=Lax`;
    return new Response(null, { status: 302, headers: { Location: '/', 'Set-Cookie': cookie } });
}

JWT 서명 / 검증 (jose)

// lib/auth/jwt.ts
const secret = new TextEncoder().encode(process.env.AUTH_SECRET);

// 서명 (로그인 시)
export async function signSession(payload: SessionPayload): Promise<string> {
    return new SignJWT({ ...payload })
        .setProtectedHeader({ alg: 'HS256' })
        .setExpirationTime('7d')
        .sign(secret);
}

// 검증 (요청마다)
export async function verifySession(token: string): Promise<SessionPayload> {
    const { payload } = await jwtVerify(token, secret);
    return payload as SessionPayload;
}
next-auth를 제거한 이유
next-auth v5 beta에서 OAuthCallback 에러가 반복 발생. 직접 GitHub OAuth 흐름을 구현해서 해결.
code → access_token → user info → JWT 쿠키 전 과정을 직접 제어.

세션 인터페이스

interface SessionPayload {
    id: number;
    name: string;
    email: string | null;
    image: string;     // GitHub 프로필 이미지 URL
    isAdmin: boolean;  // GIT_ADMIN_ID와 일치 여부
}

export const getPosts = async (page: number, limit: number, query = '') => {

export const deletePost = async (id: string) => {
