3. DB 스키마 (Oracle posts 테이블)

Oracle SQL — posts 테이블 & SEQUENCE
-- 시퀀스: id 자동 증가
CREATE SEQUENCE posts_seq START WITH 1 INCREMENT BY 1;

-- posts 테이블
CREATE TABLE posts (
    id         NUMBER        DEFAULT posts_seq.NEXTVAL PRIMARY KEY,
    title      VARCHAR2(200) NOT NULL,
    content    CLOB,
    category   VARCHAR2(50),
    author     VARCHAR2(100),
    created_at DATE          DEFAULT SYSDATE,
    updated_at DATE
);
컬럼타입설명
idNUMBERPK, posts_seq.NEXTVAL 자동 할당
titleVARCHAR2(200)게시물 제목 (NOT NULL)
contentCLOB게시물 본문 (대용량 텍스트)
categoryVARCHAR2(50)카테고리
authorVARCHAR2(100)작성자 (principal.getName())
created_atDATE작성일 (SYSDATE 기본값)
updated_atDATE수정일 (UPDATE 시 SYSDATE 세팅)
underscore → camelCase 자동 매핑
mybatis.configuration.map-underscore-to-camel-case=true 설정 덕분에
DB 컬럼 created_at → PostDto 필드 createdAt 으로 자동 매핑됨.

6. PostMapper.java & PostMapper.xml

src/main/java/com/madayV2/blog/mapper/PostMapper.java
package com.madayV2.blog.mapper;

import java.util.List;
import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;
import com.madayV2.blog.dto.PostDto;

// @Mapper: MyBatis가 이 인터페이스의 구현체를 자동 생성 & Spring Bean 등록
// 개발자가 직접 JDBC 코드 작성 불필요 — PostMapper.xml의 SQL을 읽어서 실행
@Mapper
public interface PostMapper {
    List<PostDto> findAll();

    PostDto findById(Long id);

    void insert(PostDto post);

    void update(PostDto post);

    void delete(Long id);

    // 페이지네이션: offset = page * size, limit = size
    List<PostDto> findPage(@Param("offset") int offset, @Param("limit") int limit);

    int count();
}
src/main/resources/mapper/PostMapper.xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
    "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

<mapper namespace="com.madayV2.blog.mapper.PostMapper">

    <select id="findAll" resultType="PostDto">
        SELECT id, title, content, category, author,
               created_at, updated_at
        FROM posts
        ORDER BY created_at DESC
    </select>

    <select id="findById" parameterType="Long" resultType="PostDto">
        SELECT * FROM posts WHERE id = #{id}
    </select>

    <insert id="insert" parameterType="PostDto">
        INSERT INTO posts (title, content, category, author)
        VALUES (#{title}, #{content}, #{category}, #{author})
    </insert>

    <update id="update" parameterType="PostDto">
        UPDATE posts
        SET title = #{title},
            content = #{content},
            category = #{category},
            updated_at = SYSDATE
        WHERE id = #{id}
    </update>

    <delete id="delete" parameterType="Long">
        DELETE FROM posts WHERE id = #{id}
    </delete>

    <select id="findPage" resultType="PostDto">
        SELECT id, title, content, category, author, created_at, updated_at
        FROM posts ORDER BY created_at DESC
        OFFSET #{offset} ROWS FETCH NEXT #{limit} ROWS ONLY
    </select>

    <select id="count" resultType="int">
        SELECT COUNT(*) FROM posts
    </select>

</mapper>
namespace — 이 XML이 어떤 @Mapper 인터페이스와 연결되는지 지정. 정확히 일치해야 함.
resultType="PostDto" — mybatis.type-aliases-package 설정으로 전체 경로 생략 가능.
#{} — PreparedStatement 파라미터 바인딩 → SQL 인젝션 방지. 항상 ${}가 아닌 #{}를 써야 함.
Oracle OFFSET/FETCH — Oracle 12c+ 표준 페이지네이션 문법. ROWNUM 방식보다 간결.

7. PostService

src/main/java/com/madayV2/blog/service/PostService.java
package com.madayV2.blog.service;

import java.util.List;
import org.springframework.stereotype.Service;
import com.madayV2.blog.dto.PostDto;
import com.madayV2.blog.mapper.PostMapper;
import lombok.RequiredArgsConstructor;

// @RequiredArgsConstructor: final 필드의 생성자를 자동 생성 (스프링 권장 DI 방식)
@Service
@RequiredArgsConstructor
public class PostService {
    private final PostMapper postMapper;

    public List<PostDto> getAll() {
        return postMapper.findAll();
    }

    public PostDto getById(Long id) {
        return postMapper.findById(id);
    }

    public void create(PostDto post) {
        postMapper.insert(post);
    }

    public void modify(PostDto post) {
        postMapper.update(post);
    }

    public void remove(Long id) {
        postMapper.delete(id);
    }

    // page는 0부터 시작 → offset = page * size
    public List<PostDto> getPage(int page, int size) {
        return postMapper.findPage(page * size, size);
    }

    public int getTotalPages(int size) {
        int total = postMapper.count();
        return (int) Math.ceil((double) total / size);
    }
}
@RequiredArgsConstructor vs @Autowired
@Autowired 필드 주입은 스프링 공식 문서에서 권장하지 않음. 테스트 시 모킹이 어렵고 순환 의존 감지가 늦음.
@RequiredArgsConstructor + final 필드 = 생성자 주입 방식. Spring이 권장하는 표준 패턴.

8. PostController

src/main/java/com/madayV2/blog/controller/PostController.java
package com.madayV2.blog.controller;

import java.util.List;
import java.security.Principal;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import com.madayV2.blog.dto.PostDto;
import com.madayV2.blog.service.PostService;
import lombok.RequiredArgsConstructor;

@Controller
@RequiredArgsConstructor
@RequestMapping("/posts")
public class PostController {

    private final PostService postService;

    // GET /posts → templates/posts/list.html
    @GetMapping
    public String list(Model model) {
        List<PostDto> posts = postService.getAll();
        model.addAttribute("posts", posts);
        return "posts/list";
    }

    // GET /posts/{id} → templates/posts/detail.html
    @GetMapping("/{id}")
    public String detail(@PathVariable Long id, Model model) {
        model.addAttribute("post", postService.getById(id));
        return "posts/detail";
    }

    // GET /posts/{id}/data → JSON 반환 (대시보드 모달 Ajax용)
    @GetMapping("/{id}/data")
    @ResponseBody
    public PostDto data(@PathVariable Long id) {
        return postService.getById(id);
    }

    // GET /posts/write → 글 작성 폼
    @GetMapping("/write")
    public String writeForm(Model model) {
        model.addAttribute("post", new PostDto());
        return "posts/write";
    }

    // POST /posts/write → DB 저장 후 목록으로 redirect (PRG 패턴)
    @PostMapping("/write")
    public String write(@ModelAttribute PostDto post, Principal principal) {
        post.setAuthor(principal.getName()); // 로그인 유저명 세팅
        postService.create(post);
        return "redirect:/posts";
    }

    // GET /posts/edit/{id} → 글 수정 폼
    @GetMapping("/edit/{id}")
    public String editForm(@PathVariable Long id, Model model) {
        model.addAttribute("post", postService.getById(id));
        return "posts/edit";
    }

    // POST /posts/edit/{id} → 수정 저장
    @PostMapping("/edit/{id}")
    public String edit(@PathVariable Long id, @ModelAttribute PostDto post) {
        post.setId(id);
        postService.modify(post);
        return "redirect:/posts";
    }

    // POST /posts/delete/{id} → 삭제
    @PostMapping("/delete/{id}")
    public String delete(@PathVariable Long id) {
        postService.remove(id);
        return "redirect:/posts";
    }
}
HTTP경로설명인증
GET/posts목록 조회누구나
GET/posts/{id}상세 조회누구나
GET/posts/{id}/dataJSON 반환 (Ajax)누구나
GET/posts/write작성 폼로그인 필요
POST/posts/write저장로그인 필요
GET/posts/edit/{id}수정 폼로그인 필요
POST/posts/edit/{id}수정 저장로그인 필요
POST/posts/delete/{id}삭제로그인 필요

9. DashboardController

src/main/java/com/madayV2/blog/controller/DashboardController.java
package com.madayV2.blog.controller;

import java.util.Map;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import com.madayV2.blog.service.PostService;
import com.madayV2.blog.service.SqlService;
import lombok.RequiredArgsConstructor;

@Controller
@RequiredArgsConstructor
public class DashboardController {
    private final PostService postService;
    private final SqlService sqlService;

    // GET / → 대시보드 (페이지네이션 포함)
    @GetMapping("/")
    public String dashboard(Model model,
            @RequestParam(defaultValue = "0") int page) {
        int size = 8;
        model.addAttribute("posts", postService.getPage(page, size));
        model.addAttribute("totalPages", postService.getTotalPages(size));
        model.addAttribute("currentPage", page);
        return "dashboard"; // templates/dashboard.html
    }

    // POST /sql/execute → SQL 콘솔 실행 결과 JSON 반환
    @PostMapping("/sql/execute")
    @ResponseBody
    public Map<String, Object> execute(@RequestBody Map<String, String> body) {
        try {
            return sqlService.execute(body.get("sql"));
        } catch (Exception e) {
            return Map.of("type", "error", "message", e.getMessage());
        }
    }
}
페이지네이션 흐름
GET /?page=0 → postService.getPage(0, 8) → findPage(offset=0, limit=8)
GET /?page=1 → postService.getPage(1, 8) → findPage(offset=8, limit=8)
totalPages = Math.ceil(전체건수 / 8)

10. GlobalModelAdvice

src/main/java/com/madayV2/blog/config/GlobalModelAdvice.java
package com.madayV2.blog.config;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.web.bind.annotation.ControllerAdvice;
import org.springframework.web.bind.annotation.ModelAttribute;

@ControllerAdvice
public class GlobalModelAdvice {

    @ModelAttribute("ctx")
    public String contextPath(HttpServletRequest request) {
        return request.getContextPath();
    }
}
ctx 주입 이유
context-path=/v2 설정 때문에 JS에서 fetch URL을 만들 때 접두사가 필요함.
@ControllerAdvice + @ModelAttribute("ctx") → 모든 템플릿에서 ${ctx}로 접근 가능.
Thymeleaf inline JavaScript에서 const contextPath = /*[[${ctx}]]*/ ''; 로 JS에 주입.
예: fetch(`${contextPath}/posts/${id}/data`) → 실제로 /v2/posts/1/data 로 요청됨.

12. SecurityConfig

src/main/java/com/madayV2/blog/config/SecurityConfig.java
package com.madayV2.blog.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.util.matcher.AntPathRequestMatcher;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .authorizeHttpRequests(auth -> auth
                // 누구나 접근 가능한 경로
                .requestMatchers("/", "/posts", "/posts/{id}", "/posts/{id}/data").permitAll()
                .requestMatchers("/css/**").permitAll()
                // SQL 콘솔은 인증 필요
                .requestMatchers("/sql/**").authenticated()
                // 나머지 (write, edit, delete 등) 모두 인증 필요
                .anyRequest().authenticated()
            )
            .formLogin(form -> form
                .loginPage("/login")
                .defaultSuccessUrl("/", true)
                .permitAll()
            )
            .exceptionHandling(ex -> ex
                // /sql/** 비인증 Ajax 요청 → 로그인 페이지 redirect 대신 401 직접 반환
                .defaultAuthenticationEntryPointFor(
                    (req, res, e) -> res.sendError(401, "Unauthorized"),
                    new AntPathRequestMatcher("/sql/**")
                )
            )
            .logout(logout -> logout.logoutSuccessUrl("/").permitAll());
        return http.build();
    }

    @Bean
    public UserDetailsService userDetailsService() {
        // InMemoryUserDetailsManager: DB 없이 메모리에 유저 저장
        var user = User.builder()
            .username("admin")
            .password(passwordEncoder().encode("admin1234"))
            .roles("ADMIN")
            .build();
        return new InMemoryUserDetailsManager(user);
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();  // 단방향 해시 (복호화 불가)
    }
}
경로접근 권한
/누구나 (permitAll)
/posts누구나 (permitAll)
/posts/{id}누구나 (permitAll)
/posts/{id}/data누구나 (permitAll)
/css/**누구나 (permitAll)
/sql/**인증 필요 (비인증 Ajax 요청 → 401)
/login누구나 (formLogin.permitAll)
그 외 모든 경로인증 필요 (anyRequest().authenticated)
로그인 계정: username=admin / password=admin1234
비밀번호는 BCryptPasswordEncoder로 인코딩되어 메모리에 저장됨.

package com.madayV2.blog.service;

import java.util.List;
import java.util.Map;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
public class SqlService {
    // JdbcTemplate: Spring 제공 JDBC 헬퍼
    // 커넥션 열기/닫기, 예외 변환(SQLException → DataAccessException) 자동 처리
    private final JdbcTemplate jdbcTemplate;

    public Map<String, Object> execute(String sql) {
        String upper = sql.trim().toUpperCase();

        // 조회 쿼리: SELECT, SHOW, DESC(RIBE), WITH (CTE)
        if (upper.startsWith("SELECT") || upper.startsWith("SHOW")
                || upper.startsWith("DESC") || upper.startsWith("WITH")) {
            // queryForList: 결과 → List<Map<컬럼명, 값>>
            List<Map<String, Object>> rows = jdbcTemplate.queryForList(sql.trim());
            return Map.of("type", "select", "rows", rows, "count", rows.size());

        } else if (upper.startsWith("INSERT") || upper.startsWith("UPDATE")
                || upper.startsWith("DELETE")) {
            // update(): DML 실행, 영향받은 행 수 반환
            int affected = jdbcTemplate.update(sql.trim());
            return Map.of("type", "dml", "affected", affected);

        } else {
            // execute(): CREATE TABLE, DROP, ALTER 등 DDL
            jdbcTemplate.execute(sql.trim());
            return Map.of("type", "ddl", "message", "실행 완료");
        }
    }
}
SQL 시작 키워드처리 방식반환 JSON
SELECT, SHOW, DESC, WITHqueryForList(){"type":"select","rows":[...],"count":N}
INSERT, UPDATE, DELETEupdate(){"type":"dml","affected":N}
CREATE, DROP, ALTER 등execute(){"type":"ddl","message":"실행 완료"}

왜 서버에서 직접 빌드?
GitHub Actions runner: x86_64 아키텍처. OCI 서버: ARM (aarch64).
x86에서 빌드한 Docker 이미지는 ARM에서 실행 불가 (아키텍처 불일치).
→ CI에서는 SSH로 서버에 접속만 하고, 실제 docker 빌드는 ARM 서버에서 수행.
docker-compose V1 vs V2
이 프로젝트는 docker-compose (하이픈 있음, V1 단독 설치) 사용.
V2는 docker compose (하이픈 없음, Docker CLI 플러그인 방식).
